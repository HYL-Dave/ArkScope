function declarations(file, source, exportedOnly) {
  const sf = ts.createSourceFile(file, source, ts.ScriptTarget.Latest, true);
  return sf.statements.flatMap(node => {
    if (exportedOnly && !node.modifiers?.some(x => x.kind === ts.SyntaxKind.ExportKeyword)) return [];
    const names = ts.isVariableStatement(node) ? node.declarationList.declarations.map(d => d.name) : [node.name];
    return names.filter(name => name && ts.isIdentifier(name)).map(name => name.text);
  });
}

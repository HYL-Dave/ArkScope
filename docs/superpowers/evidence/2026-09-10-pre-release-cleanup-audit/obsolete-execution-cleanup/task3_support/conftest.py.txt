"""Task 3 evidence hooks; loaded only by the approved offline launcher."""

import json
import os
from pathlib import Path
import sys
import tempfile
from urllib.parse import unquote


WORK = Path(os.environ["ARKSCOPE_OFFLINE_TEST_WORKSPACE"]).resolve()
ROOT = Path(__file__).resolve().parents[4]
MAIN = Path("/mnt/md0/PycharmProjects/ArkScope")
DENIED = []
DESELECTED = []
TEMP = WORK / "tmp"
TEMP.mkdir(parents=True, exist_ok=True)
tempfile.tempdir = str(TEMP)


def _audit(event, args):
    if event in {"open", "sqlite3.connect"} and args:
        raw = args[0]
        if isinstance(raw, (str, bytes, os.PathLike)):
            value = os.fsdecode(raw)
            if value.startswith("file:"):
                value = unquote(value[5:].split("?", 1)[0])
            path = Path(value).resolve()
            if not path.is_relative_to(WORK) and (
                path.is_relative_to(MAIN)
                or path.is_relative_to(ROOT / "data")
                or path.is_relative_to(ROOT / "config")
                or path.name == ".env"
            ):
                DENIED.append({"event": event, "path": str(path)})
                raise PermissionError("Task 3 rejected non-fixture data/config access")
    if event in {
        "socket.connect", "socket.sendto", "socket.getaddrinfo",
        "socket.gethostbyname", "socket.gethostbyaddr",
    }:
        DENIED.append({"event": event})
        raise PermissionError("Task 3 rejected network access")


sys.addaudithook(_audit)


def pytest_configure(config):
    from src.agents import config as agent_config
    from src.monitor import engine

    agent_config._MAIN_CONFIG_PATH = WORK / "synthetic-user-profile.yaml"
    agent_config._LOCAL_CONFIG_PATH = WORK / "synthetic-user-profile.local.yaml"
    engine._CONFIG_PATH = WORK / "synthetic-user-profile.yaml"
    engine._LOCAL_CONFIG_PATH = WORK / "synthetic-user-profile.local.yaml"
    agent_config.get_agent_config.cache_clear()


def pytest_deselected(items):
    DESELECTED.extend(item.nodeid for item in items)


def pytest_collection_finish(session):
    (WORK / "nodes.json").write_text(
        json.dumps([item.nodeid for item in session.items], indent=2) + "\n",
        encoding="utf-8",
    )
    (WORK / "deselected.json").write_text(
        json.dumps(DESELECTED, indent=2) + "\n", encoding="utf-8",
    )


def pytest_sessionfinish(session, exitstatus):
    (WORK / "boundary.json").write_text(
        json.dumps({"exitstatus": int(exitstatus), "denied": DENIED}, indent=2)
        + "\n", encoding="utf-8",
    )

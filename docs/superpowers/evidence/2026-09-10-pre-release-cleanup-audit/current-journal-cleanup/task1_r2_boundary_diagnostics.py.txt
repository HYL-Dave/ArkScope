"""Acknowledgement uncertainty at the durable completed-observation boundary."""

import json
import sqlite3
import time

import pytest

from src.lifecycle_investigation.agent import run_agent
from tests.lifecycle_investigation_fixtures import controller
from tests.test_lifecycle_investigation_agent import choose, completed
from tests.test_lifecycle_investigation_findings import payload


@pytest.mark.parametrize("failure", ["none", "before_commit", "after_commit"])
def test_completed_observation_ack_failure_remains_readable(tmp_path, monkeypatch, failure):
    dispatched = []

    async def model(call, credential, control):
        material = json.loads(call.prompt.split("\nMATERIAL\n")[1])
        reply = completed(call, control, choose("conclude", finding=payload(material["sources"][0]["passages"])))
        dispatched.append(call.call_id)
        return reply

    async def runner(*args, **kwargs):
        return await run_agent(*args, **kwargs, model=model)

    service, store, loads, binding = controller(tmp_path, runner=runner)
    original = store.step

    def step(*args, **kwargs):
        if kwargs["kind"] == "model_result" and failure == "before_commit":
            raise ValueError("investigation_recording_unavailable")
        original(*args, **kwargs)
        if kwargs["kind"] == "model_result" and failure == "after_commit":
            raise ValueError("investigation_recording_unavailable")

    monkeypatch.setattr(store, "step", step)
    try:
        identity = service.start(binding=binding, request_key="r2-boundary")["run_id"]
        until = time.monotonic() + 5
        while service.is_local_running(identity) and time.monotonic() < until:
            time.sleep(0.01)
        assert not service.is_local_running(identity)
        with sqlite3.connect(store.path.as_uri() + "?mode=ro", uri=True) as conn:
            status = conn.execute("SELECT status FROM lifecycle_investigation_jobs WHERE run_id=?", (identity,)).fetchone()[0]
            calls = conn.execute("SELECT call_id,terminal FROM lifecycle_investigation_calls WHERE run_id=?", (identity,)).fetchall()
            results = conn.execute("SELECT payload_json FROM lifecycle_investigation_steps WHERE run_id=? AND kind='model_result'", (identity,)).fetchall()
            saved = json.loads(conn.execute("SELECT payload_json FROM lifecycle_investigation_results WHERE run_id=?", (identity,)).fetchone()[0])
        assert status == ("succeeded" if failure == "none" else "failed")
        assert dispatched == ["step-1"] and len(loads) == 1
        assert calls == [("step-1", "completed")]
        assert len(results) == (0 if failure == "before_commit" else 1)
        assert saved["stats"]["model_submissions"] == 1
        if results:
            assert json.loads(results[0][0])["usage"] == {"input_tokens": 10, "output_tokens": 20}
        assert (saved["stats"]["input_tokens"], saved["stats"]["output_tokens"]) == ((10, 20) if failure == "none" else (None, None))
        assert service.read(identity)["status"] == status
    finally:
        service.close()
